24 /**
25 * @brief Executes a Global Greedy Resource Allocation based on surrogate pseudo-utility.
26 * @details Normalizes device resource demands against total system capacity to compute
27 * a multidimensional cost-benefit pseudo-utility and sorts candidates once globally.
28 * Allocates devices greedily per server using prefix-skipping and minimum-demand early termination.
29 *
30 * @param state The unified simulation result state containing devices, servers, and metrics.
31 */
32 inline void greedy_cost_benefit(metrics::Result& state) {
33 auto metrics_ptr = std::dynamic_pointer_cast<metrics::HeuristicMetrics>(state.stats);
34 if (!metrics_ptr) {
35 std::cerr << "Error: Invalid metrics type for heuristic solver.\n";
36 return;
37 }
38
39 core::Devices& devices = state.devices;
40 core::Servers& servers = state.servers;
41
42 auto start_chrono = std::chrono::high_resolution_clock::now();
43
44 double total_cpu = 0.0, total_mem = 0.0, total_sto = 0.0;
45 for (size_t i = 1; i < servers.size(); i++) {
46 total_cpu += servers[i].cpu;
47 total_mem += servers[i].mem;
48 total_sto += servers[i].sto;
49 }
50
51 id_pair_vector candidates;
52 candidates.reserve(devices.size());
53
54 double min_cpu = std::numeric_limits<double>::max();
55 double min_mem = std::numeric_limits<double>::max();
56 double min_sto = std::numeric_limits<double>::max();
57
58 for (size_t d = 1; d < devices.size(); d++) {
59 core::Device& device = devices[d];
60 if (!device.served) {
61 double w_cpu = device.cpu / total_cpu;
62 double w_mem = device.mem / total_mem;
63 double w_sto = device.sto / total_sto;
64
65 double utility = device.cost / (w_cpu + w_mem + w_sto);
66 candidates.emplace_back(device.id, utility);
67
68 if (device.cpu < min_cpu) min_cpu = device.cpu;
69 if (device.mem < min_mem) min_mem = device.mem;
70 if (device.sto < min_sto) min_sto = device.sto;
71 }
72 }
73
74 std::sort(candidates.begin(), candidates.end(), [](const id_pair& a, const id_pair& b) {
75 return a.second > b.second;
76 });
77
78 size_t start_idx = 0;
79 const size_t num_candidates = candidates.size();
80
81 for (size_t i = 1; i < servers.size() && start_idx < num_candidates; i++) {
82 core::Server& server = servers[i];
83
84 for (size_t c = start_idx; c < num_candidates; c++) {
85 if ((server.cpu - server.supply.cpu < min_cpu) ||
86 (server.mem - server.supply.mem < min_mem) ||
87 (server.sto - server.supply.sto < min_sto)) {
88 break;
89 }
90
91 core::Device& device = devices[candidates[c].first];
92 if (!device.served && server.can_serve(device)) {
93 server.add_service(device);
94 }
95
96 if (c == start_idx && device.served) {
97 start_idx++;
98 }
99 }
100 }
101
102 auto end_chrono = std::chrono::high_resolution_clock::now();
103 metrics_ptr->outputs.execution_time = std::chrono::duration<double>(end_chrono - start_chrono).count();
104 }

106 /**
107 * @brief Executes a fast random baseline resource allocation.
108 * @details Iterates through all devices and attempts to assign each one to a
109 * single uniformly selected server. If the selected server lacks capacity,
110 * the device remains unserved. Metric calculation is deferred.
111 *
112 * @param state The unified simulation result state containing devices, servers, and metrics.
113 */
114 inline void greedy_random(metrics::Result& state) {
115 auto metrics_ptr = std::dynamic_pointer_cast<metrics::HeuristicMetrics>(state.stats);
116 if (!metrics_ptr) {
117 std::cerr << "Error: Invalid metrics type for heuristic solver.\n";
118 return;
119 }
120
121 core::Devices& devices = state.devices;
122 core::Servers& servers = state.servers;
123
124 int size_servers = static_cast<int>(servers.size()) - 1;
125
126 auto start_chrono = std::chrono::high_resolution_clock::now();
127
128 for (size_t d = 1; d < devices.size(); d++) {
129 core::Device& device = devices[d];
130
131 if (size_servers > 1) {
132 int ec_id = randomness::random_number<int>(1, size_servers);
133 if (servers[ec_id].can_serve(device)) {
134 servers[ec_id].add_service(device);
135 }
136 }
137 }
138
139 auto end_chrono = std::chrono::high_resolution_clock::now();
140 metrics_ptr->outputs.execution_time = std::chrono::duration<double>(end_chrono - start_chrono).count();
141 }

143 /**
144 * @brief Executes a two-stage random fallback resource allocation.
145 * @details Attempts to assign each device to a randomly selected Edge (EC) server first.
146 * If the selected Edge server lacks capacity, falls back to a randomly selected Cloud (CC) server.
147 *
148 * @param state The unified simulation result state containing devices, servers, and metrics.
149 */
150 inline void greedy_random_fallback(metrics::Result& state) {
151 auto metrics_ptr = std::dynamic_pointer_cast<metrics::HeuristicMetrics>(state.stats);
152 if (!metrics_ptr) {
153 std::cerr << "Error: Invalid metrics type for heuristic solver.\n";
154 return;
155 }
156
157 int ec_count = metrics_ptr->inputs.servers_ec_count;
158 int cc_count = metrics_ptr->inputs.servers_cc_count;
159
160 core::Devices& devices = state.devices;
161 core::Servers& servers = state.servers;
162
163 auto start_chrono = std::chrono::high_resolution_clock::now();
164
165 for (size_t d = 1; d < devices.size(); d++) {
166 core::Device& device = devices[d];
167
168 if (ec_count > 0) {
169 int ec_id = randomness::random_number<int>(1, ec_count);
170 if (servers[ec_id].can_serve(device)) {
171 servers[ec_id].add_service(device);
172 continue;
173 }
174 }
175
176 if (cc_count > 0) {
177 int cc_id = randomness::random_number<int>(ec_count + 1, ec_count + cc_count);
178 if (servers[cc_id].can_serve(device)) {
179 servers[cc_id].add_service(device);
180 }
181 }
182 }
183
184 auto end_chrono = std::chrono::high_resolution_clock::now();
185 metrics_ptr->outputs.execution_time = std::chrono::duration<double>(end_chrono - start_chrono).count();
186 }
187 }

189 /**
190 * @brief Bootstraps and executes the configured constructive heuristic.
191 * @details Reads the specified algorithm identifier from `state.stats` and dispatches
192 * execution to the corresponding heuristic solver.
193 *
194 * @param state The unified simulation state to be processed.
195 */
196 inline void boot(metrics::Result& state) {
197 if (!state.stats) return;
198
199 if (metrics::Algorithm::greedy_cost_benefit == state.stats->algorithm) {
200 greedy_cost_benefit(state);
201 } else if (metrics::Algorithm::greedy_random == state.stats->algorithm) {
202 greedy_random(state);
203 } else if (metrics::Algorithm::greedy_random_fallback == state.stats->algorithm) {
204 greedy_random_fallback(state);
205 } else {
206 std::cerr << "Error: Unknown heuristic algorithm type.\n";
207 }
208 }
