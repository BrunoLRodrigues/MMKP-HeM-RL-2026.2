31 /**
32 * @brief Formulates and solves the exact ILP resource allocation model using CPLEX.
33 * @details Defines binary decision variables \f$x_{d}^{i} \in \{0, 1\}\f$ indicating whether
34 * device \f$d\f$ is assigned to server \f$i\f$. Maximizes the total utility of served devices
35 * (avoided SLA penalties) subject to single-server assignment and multidimensional
36 * resource capacity constraints (CPU, MEM, STO).
37 *
38 * @param state The unified simulation state containing devices, servers, and metrics.
39 * @param time_limit_sec The maximum execution time allowed for the solver in seconds.
40 */
41 inline void solver_maximum_cost(metrics::Result& state, int time_limit_sec = 3600) {
42 core::Devices& devices = state.devices;
43 core::Servers& servers = state.servers;
44
45 auto metrics_ptr = std::dynamic_pointer_cast<metrics::ExactMetrics>(state.stats);
46 if (!metrics_ptr) {
47 std::cerr << "Error: Invalid metrics type for exact solver.\n";
48 return;
49 }
50
51 IloEnv env;
52 try {
53 IloModel model(env);
54 const size_t num_devices = devices.size();
55 const size_t num_servers = servers.size();
56
57 #pragma region VariableDeclaration
58 //=========================================================================
59 // 1. VARIABLE DECLARATION
60 //=========================================================================
61
62 // x^{i}_{d}: 1 if device d is allocated to server i, 0 otherwise.
63 NumVar2D x(env, num_servers);
64 for (size_t i = 1; i < num_servers; i++) {
65 x[i] = IloNumVarArray(env, num_devices, 0, 1, ILOBOOL);
66 }
67 #pragma endregion
68
69 #pragma region ObjectiveFunctionAndConstraints
70 IloExpr obj(env);
71
72 for (size_t d = 1; d < num_devices; d++) {
73 IloExpr c1(env);
74 for (size_t i = 1; i < num_servers; i++) {
75 c1 += x[i][d];
76 obj += devices[d].cost * x[i][d];
77 }
78 model.add(c1 <= 1);
79 c1.end();
80 }
81
82 model.add(IloMaximize(env, obj));
83 obj.end();
84
85 for (size_t i = 1; i < num_servers; i++) {
86 IloExpr c_cpu(env), c_mem(env), c_sto(env);
87
88 for (size_t d = 1; d < num_devices; d++) {
89 c_cpu += devices[d].cpu * x[i][d];
90 c_mem += devices[d].mem * x[i][d];
91 c_sto += devices[d].sto * x[i][d];
92 }
93
94 model.add(c_cpu <= servers[i].cpu);
95 model.add(c_mem <= servers[i].mem);
96 model.add(c_sto <= servers[i].sto);
97
98 c_cpu.end();
99 c_mem.end();
100 c_sto.end();
101 }
102 #pragma endregion
103
104 #pragma region SolverConfiguration
105 //=========================================================================
106 // 4. SOLVER CONFIGURATION AND EXECUTION
107 //=========================================================================
108
109 IloCplex cplex(model);
110 cplex.setOut(env.getNullStream());
111 cplex.setWarning(env.getNullStream());
112
113 cplex.setParam(IloCplex::Param::Threads, 1);
114 cplex.setParam(IloCplex::Param::ClockType, 2);
115 cplex.setParam(IloCplex::Param::TimeLimit, time_limit_sec);
116
117 cplex.setParam(IloCplex::Param::WorkMem, 16384);
118 cplex.setParam(IloCplex::Param::MIP::Strategy::File, 3);
119
120 // Relative MIP gap tolerance of 1e-3 (0.001)
121 cplex.setParam(IloCplex::Param::MIP::Tolerances::MIPGap, 1e-3);
122 cplex.setParam(IloCplex::Param::Emphasis::MIP, IloCplex::MIPEmphasisBalanced);
123
124 auto start_chrono = std::chrono::high_resolution_clock::now();
125 const IloBool has_solution = cplex.solve();
126 auto end_chrono = std::chrono::high_resolution_clock::now();
127
128 metrics_ptr->outputs.execution_time = std::chrono::duration<double>(end_chrono - start_chrono).count();
129
130 std::stringstream status;
131 status << cplex.getStatus();
132 metrics_ptr->status = status.str();
133 #pragma endregion
134
135 #pragma region Result
136 //=========================================================================
137 // 5. PARSE RESULTS
138 //=========================================================================
139
140 if (has_solution) {
141 metrics_ptr->objective_function = static_cast<double>(cplex.getObjValue());
142 metrics_ptr->gap = static_cast<double>(cplex.getMIPRelativeGap());
143
144 for (size_t i = 1; i < num_servers; i++) {
145 for (size_t d = 1; d < num_devices; d++) {
146 if (cplex.isExtracted(x[i][d]) && cplex.getValue(x[i][d]) > 0.5) {
147 servers[i].add_service(devices[d]);
148 }
149 }
150 }
151 }
152
153 } catch (const IloException& e) {
154 std::cerr << "CPLEX Error: " << e.getMessage() << "\n";
155 } catch (const std::exception& e) {
156 std::cerr << "Standard Exception: " << e.what() << "\n";
157 } catch (...) {
158 std::cerr << "An unknown error occurred in the CPLEX model.\n";
159 }
160 env.end();
161 }
162 }
163
164 /**
165 * @brief Bootstraps and executes the configured exact mathematical model.
166 * @details Reads the specified algorithm identifier from `state.stats` and dispatches
167 * execution to the corresponding exact solver implementation.
168 *
169 * @param state The unified simulation state to be processed.
170 * @param time_limit_sec The maximum execution time allowed for the exact solver in seconds.
171 */
172 inline void boot(metrics::Result& state, int time_limit_sec = 3600) {
173 if (!state.stats) return;
174
175 if (metrics::Algorithm::solver_maximum_cost == state.stats->algorithm) {
176 solver_maximum_cost(state, time_limit_sec);
177 } else {
178 std::cerr << "Error: Unknown mathematical model algorithm type.\n";
179 }
180 }
