26/**
27 * @brief Executes a specific algorithm configuration for the network resource allocation simulation.
28 * @details Instantiates the appropriate metrics tracker based on the selected option, dispatches
29 * the simulation state to the corresponding solver paradigm (Exact, Heuristic, or Meta-heuristic),
30 * computes final utilization/cost metrics, and saves the results to disk.
31 *
32 * @param base The initial system state containing untouched devices and servers.
33 * @param option The identifier for the algorithm combination to execute:
34 * - `0`: Exact CPLEX solver (`solver_maximum_cost`)
35 * - `-1`: Constructive heuristic (`greedy_cost_benefit`)
36 * - `-2`: Constructive heuristic (`greedy_random`)
37 * - `-3`: Constructive heuristic (`greedy_random_fallback`)
38 * - `1`: Simulated Annealing + `greedy_cost_benefit`
39 * - `2`: Simulated Annealing + `greedy_random`
40 * - `3`: Simulated Annealing + `greedy_random_fallback`
41 * @param run_id The current execution run number (1-indexed).
42 */
43void run_simulation(const metrics::Result& base, int option, int run_id) {
44 metrics::Paradigm paradigm = metrics::Paradigm::NONE;
45 metrics::Algorithm algorithm = metrics::Algorithm::NONE;
46 metrics::Algorithm initial_heuristic = metrics::Algorithm::NONE;
47
48 if (option == 0) {
49 paradigm = metrics::Paradigm::EXACT;
50 algorithm = metrics::Algorithm::solver_maximum_cost;
51 } else if (option == -1) {
52 paradigm = metrics::Paradigm::HEURISTIC;
53 algorithm = metrics::Algorithm::greedy_cost_benefit;
54 } else if (option == -2) {
55 paradigm = metrics::Paradigm::HEURISTIC;
56 algorithm = metrics::Algorithm::greedy_random;
57 } else if (option == -3) {
58 paradigm = metrics::Paradigm::HEURISTIC;
59 algorithm = metrics::Algorithm::greedy_random_fallback;
60 } else if (option == 1) {
61 paradigm = metrics::Paradigm::METAHEURISTIC;
62 algorithm = metrics::Algorithm::simulated_annealing;
63 initial_heuristic = metrics::Algorithm::greedy_cost_benefit;
64 } else if (option == 2) {
65 paradigm = metrics::Paradigm::METAHEURISTIC;
66 algorithm = metrics::Algorithm::simulated_annealing;
67 initial_heuristic = metrics::Algorithm::greedy_random;
68 } else if (option == 3) {
69 paradigm = metrics::Paradigm::METAHEURISTIC;
70 algorithm = metrics::Algorithm::simulated_annealing;
71 initial_heuristic = metrics::Algorithm::greedy_random_fallback;
72 } else {
73 std::cerr << "Error: Invalid option selected for simulation.\n";
74 return;
75 }
76
77 metrics::Result state = base;
78
79 if (paradigm == metrics::Paradigm::EXACT) {
80 state.stats = std::make_shared<metrics::ExactMetrics>(algorithm, run_id, base.stats);
81 } else if (paradigm == metrics::Paradigm::HEURISTIC) {
82 state.stats = std::make_shared<metrics::HeuristicMetrics>(algorithm, run_id, base.stats);
83 } else if (paradigm == metrics::Paradigm::METAHEURISTIC) {
84 state.stats = std::make_shared<metrics::MetaHeuristicMetrics>(algorithm, run_id, base.stats, TEMPERATURE, ALPHA, initial_heuristic);
85 }
86
87 if (paradigm == metrics::Paradigm::EXACT) {
88 exact::boot(state, CPLEX_TIME_LIMIT);
89 } else if (paradigm == metrics::Paradigm::HEURISTIC) {
90 heuristic::boot(state);
91 } else if (paradigm == metrics::Paradigm::METAHEURISTIC) {
92 metaheuristic::boot(state);
93 }
94
95 metrics::calculate_metrics(state);
96 metrics::save_results(state, false);
97}

99/**
100 * @brief Main entry point for the network resource allocation simulation benchmark.
101 * @details Iterates over the configured range of device counts and independent runs,
102 * executing deterministic algorithms once per instance and stochastic algorithms across
103 * all runs.
104 *
105 * @return int Exit status code (`0` on normal termination).
106 */
107int main() {
108 for (int devices_length = DEVICES_START; devices_length <= DEVICES_END; devices_length += DEVICES_STEP) {
109 metrics::Result base_state(devices_length, SERVICES, EC_LENGTH, CC_LENGTH, FIXED_PENALTY);
110
111 for (int run = 1; run <= MAX_RUNS; run++) {
112 for (int option = 0; option <= 3; option++) {
113 if (run > 1 && (option == 0 || option == -1)) continue;
114 run_simulation(base_state, option, run);
115 }
116 }
117 }
118
119 return 0;
120}
