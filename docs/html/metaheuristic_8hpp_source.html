54 inline Move apply_neighbor(core::Devices& devices, core::Servers& servers, int tries = 10) {
55 Move move;
56 const int max_dev = static_cast<int>(devices.size() - 1);
57 const int max_srv = static_cast<int>(servers.size() - 1);
58
59 move.device_id = randomness::random_number<int>(1, max_dev);
60 core::Device& device = devices[move.device_id];
61 move.old_server_id = device.server_id;
62
63 int count = 0;
64 while (count < tries) {
65 int i = (!device.served && count == tries - 1)
66 ? max_srv
67 : randomness::random_number<int>(1, max_srv);
68
69 if (i == device.server_id) continue;
70 count++;
71
72 core::Server& new_server = servers[i];
73
74 if (new_server.can_serve(device)) {
75 if (device.served) {
76 servers[move.old_server_id].rmv_service(device);
77 move.delta_cost = 0.0;
78 } else {
79 move.delta_cost = -device.cost;
80 }
81
82 new_server.add_service(device);
83 move.new_server_id = i;
84 move.valid = true;
85 return move;
86 }
87 }
88
89 if (!device.served) {
90 for (int t = 0; t < tries; t++) {
91 int cand_id = randomness::random_number<int>(1, max_dev);
92 core::Device& cand = devices[cand_id];
93 if (!cand.served) continue;
94
95 int srv_id = cand.server_id;
96 core::Server& target_server = servers[srv_id];
97
98 target_server.rmv_service(cand);
99 if (target_server.can_serve(device)) {
100 target_server.add_service(device);
101 move.new_server_id = srv_id;
102 move.swapped_id = cand_id;
103 move.delta_cost = cand.cost - device.cost;
104 move.valid = true;
105 return move;
106 }
107 target_server.add_service(cand);
108 }
109 }
110
111 return move;
112 }

145 inline void simulated_annealing(metrics::Result& state) {
146 auto metrics_ptr = std::dynamic_pointer_cast<metrics::MetaHeuristicMetrics>(state.stats);
147 if (!metrics_ptr) {
148 std::cerr << "Error: Invalid metrics type for heuristic solver.\n";
149 return;
150 }
151
152 core::Devices best_devices = state.devices;
153 core::Servers best_servers = state.servers;
154
155 double best_cost = metrics_ptr->outputs.total_penalty_cost;
156 double current_cost = best_cost;
157 bool is_best_synced = true;
158
159 double T = metrics_ptr->temperature;
160 double alpha = metrics_ptr->alpha;
161
162 auto start_chrono = std::chrono::high_resolution_clock::now();
163
164 while (T > 1e-3) {
165 for (int i = 0; i < LOOP; i++) {
166 Move move = apply_neighbor(state.devices, state.servers);
167 if (!move.valid) continue;
168
169 double neighbor_cost = current_cost + move.delta_cost;
170 double delta = neighbor_cost - current_cost;
171
172 if (delta < 0.0 || (randomness::random_number(0.0, 1.0) < std::exp(-delta / T))) {
173 if (delta > 0.0 && !is_best_synced) {
174 revert_neighbor(state.devices, state.servers, move);
175 best_devices = state.devices;
176 best_servers = state.servers;
177 is_best_synced = true;
178 if (move.swapped_id != -1) {
179 state.servers[move.new_server_id].rmv_service(state.devices[move.swapped_id]);
180 }
181 if (move.old_server_id != -1) {
182 state.servers[move.old_server_id].rmv_service(state.devices[move.device_id]);
183 }
184 if (move.new_server_id != -1) {
185 state.servers[move.new_server_id].add_service(state.devices[move.device_id]);
186 }
187 }
188
189 current_cost = neighbor_cost;
190
191 if (current_cost < best_cost - ZERO) {
192 i = 0;
193 best_cost = current_cost;
194 is_best_synced = false;
195 }
196 } else {
197 revert_neighbor(state.devices, state.servers, move);
198 }
199 }
200 T *= alpha;
201 }
202
203 if (is_best_synced) {
204 state.devices = std::move(best_devices);
205 state.servers = std::move(best_servers);
206 }
207
208 auto end_chrono = std::chrono::high_resolution_clock::now();
209 metrics_ptr->outputs.execution_time += std::chrono::duration<double>(end_chrono - start_chrono).count();
210 }
211 }
