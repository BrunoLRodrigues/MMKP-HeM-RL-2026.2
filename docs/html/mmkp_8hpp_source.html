97 /**
98 * @struct Server
99 * @brief Represents an Edge Computing (EC) or Cloud Computing (CC) server node.
100 */
101 struct Server {
102 int id; ///< @brief Unique 1-based identifier of the server.
103 char type; ///< @brief Server tier type (`'E'` for Edge, `'C'` for Cloud).
104 double cpu; ///< @brief Maximum total CPU capacity of the server.
105 double mem; ///< @brief Maximum total memory capacity of the server.
106 double sto; ///< @brief Maximum total storage capacity of the server.
107 Supply supply; ///< @brief Current resource utilization and set of allocated devices.
108
109 /**
110 * @brief Default constructor initializing an empty/placeholder server.
111 */
112 Server() : id(0), type(' '), cpu(0.0), mem(0.0), sto(0.0), supply() {}
113
114 /**
115 * @brief Parameterized constructor initializing a server with given capacities.
116 * @param id_ Unique 1-based server ID.
117 * @param type_ Server tier character (`'E'` or `'C'`).
118 * @param cpu_ Total CPU capacity.
119 * @param mem_ Total memory capacity.
120 * @param sto_ Total storage capacity.
121 */
122 Server(int id_, char type_, double cpu_, double mem_, double sto_)
123 : id(id_), type(type_), cpu(cpu_), mem(mem_), sto(sto_) {}
124
125 /**
126 * @brief Checks if the server has enough residual capacity to serve a given device.
127 * @param device The candidate device to be checked.
128 * @return `true` if residual CPU, MEM, and STO capacities are sufficient; `false` otherwise.
129 */
130 inline bool can_serve(const Device& device) const {
131 return (
132 (device.cpu <= (cpu - supply.cpu)) &&
133 (device.mem <= (mem - supply.mem)) &&
134 (device.sto <= (sto - supply.sto))
135 );
136 }
137
138 /**
139 * @brief Allocates a device to this server and updates consumed resources.
140 * @param device The device to allocate (modified in-place to mark `served = true` and `server_id`).
141 * @return `true` if the device was successfully added; `false` if it was already present.
142 */
143 inline bool add_service(Device& device) {
144 auto result = supply.devices.insert(device.id);
145 if (!result.second) return false;
146
147 device.served = true;
148 device.server_id = id;
149
150 supply.cost += device.cost;
151 supply.cpu += device.cpu;
152 supply.mem += device.mem;
153 supply.sto += device.sto;
154
155 return true;
156 }
157
158 /**
159 * @brief Removes a device from this server and releases its consumed resources.
160 * @param device The device to remove (modified in-place to reset `served = false` and `server_id = -1`).
161 * @return `true` if the device was successfully removed; `false` if it was not allocated to this server.
162 */
163 inline bool rmv_service(Device& device) {
164 if (supply.devices.erase(device.id) == 0) return false;
165
166 device.served = false;
167 device.server_id = -1;
168
169 supply.cost -= device.cost;
170 supply.cpu -= device.cpu;
171 supply.mem -= device.mem;
172 supply.sto -= device.sto;
173
174 return true;
175 }
176 };

195 /**
196 * @brief Generates or loads network service requirement templates.
197 * @details Creates random service profiles with varying task counts, memory, storage,
198 * and processing demands, persisting them to `data/Services.txt`. If the file already exists,
199 * reloads the profiles and updates their penalty costs (`COST`) according to `fixed_penalty`.
200 *
201 * @param num_services The total number of distinct service profiles to generate.
202 * @param fixed_penalty If `true`, assigns a unit penalty cost (`1.0`); if `false`, generates task-proportional random penalty costs.
203 * @return A 2D string vector containing the generated or updated services dataset.
204 */
205 inline std::vector<std::vector<std::string>> services(const int num_services = 10, bool fixed_penalty = true) {
206 std::filesystem::path service_path = DATA_PATH / "Services.txt";
207
208 if (std::filesystem::exists(service_path)) {
209 auto loaded = csv::read(service_path.string(), ';').value_or(std::vector<std::vector<std::string>>{});
210
211 if (loaded.size() > 1) {
212 for (size_t i = 1; i < loaded.size(); i++) {
213 int tsk = std::stoi(loaded[i][5]);
214 double cos = 1.0;
215 if (!fixed_penalty) {
216 for (int j = 0; j < tsk; j++) {
217 cos += randomness::random_number(ZERO, 2.5);
218 }
219 }
220 loaded[i][1] = convert::to_string(cos, DECIMAL_PLACES);
221 }
222 csv::write(service_path.string(), loaded, ';');
223 }
224 return loaded;
225 }
226
227 std::vector<std::vector<std::string>> services;
228 services.push_back({"#", "COST", "CPU", "MEM", "STO", "TSK"});
229
230 for (int i = 1; i <= num_services; i++) {
231 int tsk = randomness::random_number(1, 4);
232 double cpu = 0.0, mem = 0.0, sto = 0.0, cos = 1.0;
233
234 for (int j = 0; j < tsk; j++) {
235 if (!fixed_penalty) {
236 cos += randomness::random_number(ZERO, 2.5);
237 }
238 cpu += randomness::random_number(ZERO, 2.5);
239 mem += randomness::random_number(ZERO, 2.5);
240 sto += randomness::random_number(ZERO, 15.0);
241 }
242
243 services.push_back({
244 convert::to_string(i), convert::to_string(cos, DECIMAL_PLACES),
245 convert::to_string(cpu, DECIMAL_PLACES), convert::to_string(mem, DECIMAL_PLACES), convert::to_string(sto, DECIMAL_PLACES),
246 convert::to_string(tsk)
247 });
248 }
249
250 if (!csv::write(service_path.string(), services, ';')) {
251 return {};
252 }
253 return services;
254 }

256 /**
257 * @brief Generates or loads device request data representing IoT network demands.
258 * @details Uniformly samples service requirement profiles from the services dataset
259 * and assigns them to each device instance in `data/devices/Devices_<length>.txt`. If the file
260 * already exists, synchronizes each device's penalty cost (`COST`) with its corresponding service template.
261 *
262 * @param length The total number of devices to generate.
263 * @param num_services The number of underlying service templates to sample from.
264 * @param fixed_penalty Whether the penalty (cost) of each device is fixed (`1.0`) or variable.
265 * @return A 2D string vector representing the devices dataset.
266 */
267 inline std::vector<std::vector<std::string>> devices(const int length, const int num_services = 10, bool fixed_penalty = true) {
268 std::filesystem::path device_file_path = DATA_PATH / "devices" / ("Devices_" + std::to_string(length) + ".txt");
269
270 std::vector<std::vector<std::string>> data_services = services(num_services, fixed_penalty);
271 if (data_services.size() <= 1) return {};
272
273 if (std::filesystem::exists(device_file_path)) {
274 auto loaded = csv::read(device_file_path.string(), ';').value_or(std::vector<std::vector<std::string>>{});
275
276 if (loaded.size() > 1) {
277 for (size_t i = 1; i < loaded.size(); i++) {
278 int svc = std::stoi(loaded[i][6]);
279 loaded[i][1] = data_services.at(svc).at(1);
280 }
281 csv::write(device_file_path.string(), loaded, ';');
282 }
283
284 return loaded;
285 }
286
287 std::vector<std::vector<std::string>> devices;
288 devices.push_back({"#", "COST", "CPU", "MEM", "STO", "TSK", "SVC"});
289
290 for (int i = 1; i <= length; i++) {
291 const auto& source_service_row = data_services.at(randomness::random_number(1, static_cast<int>(data_services.size()) - 1));
292
293 devices.push_back({
294 convert::to_string(i), // #
295 source_service_row.at(1), // COST
296 source_service_row.at(2), // CPU
297 source_service_row.at(3), // MEM
298 source_service_row.at(4), // STO
299 source_service_row.at(5), // TASKS
300 source_service_row.at(0) // SERVICE
301 });
302 }
303
304 if (!csv::write(device_file_path.string(), devices, ';')) {
305 return {};
306 }
307 return devices;
308 }

310 /**
311 * @brief Generates or loads server infrastructure data for Edge and Cloud nodes.
312 * @details Creates capacity profiles for `ec_length` Edge servers (`'E'`) and `cc_length`
313 * Cloud servers (`'C'`), saving the table to `data/servers/servers_<ec>_<cc>.txt`.
314 *
315 * @param ec_length The number of Edge Computing servers to generate.
316 * @param cc_length The number of Cloud Computing servers to generate.
317 * @return A 2D string vector representing the servers dataset.
318 */
319 inline std::vector<std::vector<std::string>> servers(const int ec_length = 99, const int cc_length = 1) {
320 std::filesystem::path server_file_path = DATA_PATH / "servers" / ("servers_" + std::to_string(ec_length) + "_" + std::to_string(cc_length) + ".txt");
321 if (std::filesystem::exists(server_file_path)) {
322 return csv::read(server_file_path.string(), ';').value_or(std::vector<std::vector<std::string>>{});
323 }
324
325 std::vector<std::vector<std::string>> servers;
326 servers.push_back({"#", "Type", "CPU", "MEM", "STO"});
327 int size = ec_length + cc_length;
328 double cpu, mem, sto;
329
330 for (int it = 1; it <= size; it++) {
331 if (it <= ec_length) {
332 cpu = randomness::random_number(2.5, 25.0);
333 mem = randomness::random_number(2.5, 125.0);
334 sto = randomness::random_number(15.0, 1000.0);
335 } else {
336 double factor = 2.0;
337 cpu = randomness::random_number(100000.0/factor, 200000.0/factor);
338 mem = randomness::random_number(51200.0/factor, 102800.0/factor);
339 sto = randomness::random_number(500000.0/factor, 1000000.0/factor);
340 }
341
342 servers.push_back({
343 convert::to_string(it), // IDX
344 (it <= ec_length) ? "E" : "C", // TYPE
345 convert::to_string(cpu, DECIMAL_PLACES), // CPU
346 convert::to_string(mem, DECIMAL_PLACES), // MEM
347 convert::to_string(sto, DECIMAL_PLACES) // STO
348 });
349 }
350
351 if (!csv::write(server_file_path.string(), servers, ';')) {
352 return {};
353 }
354 return servers;
355 }

925 * @struct Result
926 * @brief Encapsulates the complete system state (`Devices`, `Servers`) and polymorphic `Metrics` for a simulation run.
927 */
928 struct Result {
929 core::Devices devices; ///< @brief Vector of all IoT devices in the simulation.
930 core::Servers servers; ///< @brief Vector of all Edge and Cloud servers in the simulation.
931 std::shared_ptr<Metrics> stats; ///< @brief Polymorphic pointer to the run's metrics tracker.
932
933 /**
934 * @brief Constructs a `Result` state from pre-loaded devices and servers vectors.
935 * @param d Vector of devices.
936 * @param s Vector of servers.
937 */
938 Result(core::Devices d, core::Servers s)
939 : devices(std::move(d)), servers(std::move(s)) {}
940
941 /**
942 * @brief Deep-copy constructor cloning both entities and the polymorphic `stats` pointer.
943 * @param other Source `Result` instance to copy.
944 */
945 Result(const Result& other)
946 : devices(other.devices), servers(other.servers) {
947 if (other.stats) {
948 stats = other.stats->clone();
949 }
950 }
951
952 /**
953 * @brief Loads (or generates) an instance with the specified dimensions and initializes base metrics.
954 * @param num_devices Number of IoT devices to load.
955 * @param num_services Number of service templates.
956 * @param num_ec Number of Edge Computing servers.
957 * @param num_cc Number of Cloud Computing servers.
958 * @param fixed_penalty Whether the penalty (cost) of each device is fixed or can vary.
959 */
960 Result(int num_devices, int num_services, int num_ec, int num_cc, bool fixed_penalty = true)
961 : devices(data::load::devices(num_devices, num_services, fixed_penalty)), servers(data::load::servers(num_ec, num_cc)) {
962
963 if (devices.size() <= 1 || servers.size() <= 1) {
964 std::cerr << "Error: Data failed to load.\n";
965 exit(0);
966 }
967
968 stats = std::make_shared<Metrics>(Paradigm::NONE, Algorithm::NONE, 0);
969
970 stats->inputs.devices_count = static_cast<int>(this->devices.size()) - 1;
971 stats->inputs.servers_ec_count = num_ec;
972 stats->inputs.servers_cc_count = num_cc;
973
974 double total_cost = 0.0;
975 for (size_t d = 1; d < this->devices.size(); ++d) {
976 total_cost += this->devices[d].cost;
977 }
978 stats->inputs.max_penalty_cost = static_cast<double>(total_cost);
979 }
980
981 /**
982 * @brief Deep-copy assignment operator cloning the polymorphic `stats` object.
983 * @param other Source `Result` instance.
984 * @return Reference to `*this`.
985 */
986 Result& operator=(const Result& other) {
987 if (this != &other) {
988 devices = other.devices;
989 servers = other.servers;
990 if (other.stats) stats = other.stats->clone();
991 else stats.reset();
992 }
993 return *this;
994 }

997 /**
998 * @brief Computes the final simulation metrics based on the current device-to-server allocations.
999 * @param state The simulation result structure holding `devices`, `servers`, and `stats`.
1000 */
1001 inline void calculate_metrics(Result& state) {
1002 if (!state.stats) return;
1003
1004 auto& out = state.stats->outputs;
1005
1006 out.devices_served_count = 0;
1007 out.devices_served_ec_count = 0;
1008 out.devices_served_cc_count = 0;
1009 out.servers_used_count = 0;
1010 out.servers_used_ec_count = 0;
1011 out.servers_used_cc_count = 0;
1012 out.total_penalty_cost = 0.0;
1013
1014 for (size_t d = 1; d < state.devices.size(); ++d) {
1015 const auto& device = state.devices[d];
1016 if (device.served) {
1017 out.devices_served_count++;
1018 if (device.server_id != -1) {
1019 if (state.servers[device.server_id].type == 'E') {
1020 out.devices_served_ec_count++;
1021 } else {
1022 out.devices_served_cc_count++;
1023 }
1024 }
1025 } else {
1026 out.total_penalty_cost += device.cost;
1027 }
1028 }
1029
1030 for (size_t s = 1; s < state.servers.size(); ++s) {
1031 const auto& server = state.servers[s];
1032 if (!server.supply.devices.empty()) {
1033 out.servers_used_count++;
1034 if (server.type == 'E') {
1035 out.servers_used_ec_count++;
1036 } else {
1037 out.servers_used_cc_count++;
1038 }
1039 }
1040 }
1041 }
