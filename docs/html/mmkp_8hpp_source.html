143 inline bool add_service(Device& device) {
144 auto result = supply.devices.insert(device.id);
145 if (!result.second) return false;
146
147 device.served = true;
148 device.server_id = id;
149
150 supply.cost += device.cost;
151 supply.cpu += device.cpu;
152 supply.mem += device.mem;
153 supply.sto += device.sto;
154
155 return true;
156 }

163 inline bool rmv_service(Device& device) {
164 if (supply.devices.erase(device.id) == 0) return false;
165
166 device.served = false;
167 device.server_id = -1;
168
169 supply.cost -= device.cost;
170 supply.cpu -= device.cpu;
171 supply.mem -= device.mem;
172 supply.sto -= device.sto;
173
174 return true;
175 }

203 inline std::vector<std::vector<std::string>> services(const int num_services = 10) {
204 std::filesystem::path service_path = DATA_PATH / "Services.txt";
205 if (std::filesystem::exists(service_path)) {

207 }
208
209 std::vector<std::vector<std::string>> services;
210 services.push_back({"#", "COST", "CPU", "MEM", "STO", "TSK"});
211
212 for (int i = 1; i <= num_services; i++) {
213 int tsk = randomness::random_number(1, 4);
214 double cpu = 0.0, mem = 0.0, sto = 0.0, cos = 1.0;
215
216 for (int j = 0; j < tsk; j++) {
217 //cos += randomness::random_number(ZERO, 2.5);

223 services.push_back({
224 convert::to_string(i), convert::to_string(cos, DECIMAL_PLACES),
225 convert::to_string(cpu, DECIMAL_PLACES), convert::to_string(mem, DECIMAL_PLACES), convert::to_string(sto, DECIMAL_PLACES),
226 convert::to_string(tsk)
227 });
228 }
229
230 if (!csv::write(service_path.string(), services, ';')) {
231 return {};
232 }
233 return services;

288 inline std::vector<std::vector<std::string>> servers(const int ec_length = 99, const int cc_length = 1) {
289 std::filesystem::path server_file_path = DATA_PATH / "servers" / ("servers_" + std::to_string(ec_length) + "_" + std::to_string(cc_length) + ".txt");
290 if (std::filesystem::exists(server_file_path)) {
291 return csv::read(server_file_path.string(), ';').value_or(std::vector<std::vector<std::string>>{});
292 }
293
294 std::vector<std::vector<std::string>> servers;
295 servers.push_back({"#", "Type", "CPU", "MEM", "STO"});
296 int size = ec_length + cc_length;
297 double cpu, mem, sto;
298
299 for (int it = 1; it <= size; it++) {
300 if (it <= ec_length) {
301 cpu = randomness::random_number(2.5, 25.0);
302 mem = randomness::random_number(2.5, 125.0);
303 sto = randomness::random_number(15.0, 1000.0);
304 } else {
305 double factor = 2.0;
306 cpu = randomness::random_number(100000.0/factor, 200000.0/factor);
307 mem = randomness::random_number(51200.0/factor, 102800.0/factor);
308 sto = randomness::random_number(500000.0/factor, 1000000.0/factor);
309 }
310
311 servers.push_back({
312 convert::to_string(it), // IDX
313 (it <= ec_length) ? "E" : "C", // TYPE
314 convert::to_string(cpu, DECIMAL_PLACES), // CPU
315 convert::to_string(mem, DECIMAL_PLACES), // MEM
316 convert::to_string(sto, DECIMAL_PLACES) // STO
317 });
318 }
319
320 if (!csv::write(server_file_path.string(), servers, ';')) {
321 return {};
322 }
323 return servers;
324 }

970

972
973 out.devices_served_count = 0;
974 out.devices_served_ec_count = 0;
975 out.devices_served_cc_count = 0;
976 out.servers_used_count = 0;
977 out.servers_used_ec_count = 0;
978 out.servers_used_cc_count = 0;

982 const auto& device = state.devices[d];
983 if (device.served) {
984 out.devices_served_count++;
985 if (device.server_id != -1) {
986 if (state.servers[device.server_id].type == 'E') {
987 out.devices_served_ec_count++;
988 } else {
989 out.devices_served_cc_count++;
990 }
991 }
992 } else {
993 out.total_penalty_cost += device.cost;
994 }
995 }
996
997 for (size_t s = 1; s < state.servers.size(); ++s) {
998 const auto& server = state.servers[s];
999 if (!server.supply.devices.empty()) {
1000 out.servers_used_count++;
1001 if (server.type == 'E') {
1002 out.servers_used_ec_count++;
1003 } else {
1004 out.servers_used_cc_count++;
1005 }
