308 template<typename T>
309 inline T random_number(T min, T max, std::vector<T>& exclude) {
310 static_assert(std::is_arithmetic_v<T>, "T must be an arithmetic type.");
311
312 if (min > max) {
313 throw std::invalid_argument("Error in random_number: min cannot be greater than max.");
314 }
315
316 if constexpr (std::is_integral_v<T>) {
317 if (exclude.size() >= static_cast<size_t>(max - min + 1)) {
318 throw std::invalid_argument("Error in random_number: all possible values are excluded causing infinite loop.");
319 }
320 }
321
322 T result;
323 if constexpr (std::is_integral_v<T>) {
324 std::uniform_int_distribution<T> distribution(min, max);
325 do {
326 result = distribution(get_engine());
327 } while (std::find(exclude.begin(), exclude.end(), result) != exclude.end());
328 } else {
329 std::uniform_real_distribution<T> distribution(min, max);
330 do {
331 result = distribution(get_engine());
332 } while (std::find(exclude.begin(), exclude.end(), result) != exclude.end());
333 }
334
335 exclude.push_back(result);
336 return result;
337 }

381 template <bool Ascending, typename T, typename AttributeType, AttributeType T::* AttributePtr>
382 void sort_indices_impl(const std::vector<T>& entities, std::vector<int>& idxs) {
383 std::sort(idxs.begin(), idxs.end(), [&](int i, int j) {
384 const auto& attr_i = entities.at(i).*AttributePtr;
385 const auto& attr_j = entities.at(j).*AttributePtr;
386
387 if (attr_i != attr_j) {
388 if constexpr (Ascending) {
389 return attr_i < attr_j; // Ascending order
390 } else {
391 return attr_i > attr_j; // Descending order
392 }
393 }
394 return i < j; // Tie-breaker
395 });
396 }
397 }
